Intercropping in TZA LSMS - ISA wave 4

Starting by exploring the cereal legume intercropping in Tanzania employing LSMS-ISA data. The employed data is from wave 4 which has been collected in 2014-2015.

a) Wave 4: Long Rainy Season

setwd("D:/WORK/LEG4DEV/leg4dev")

cropdata <- read_dta("data/tza/tza_wave4/ag_sec_4a.dta") |> 
  select("y4_hhid", "plotnum", "zaocode", "zaoname", "ag4a_01", "ag4a_02", "ag4a_03", "ag4a_04", "ag4a_05", "ag4a_06") |> 
  rename(intercrop = ag4a_04) %>%
  filter(!is.na(zaocode))  # Remove rows with missing zaocode
# Number of unique households with at least one intercropped plot
length(unique(cropdata$y4_hhid[cropdata$intercrop == 1]))
[1] 1233
# Total number of intercropped plots
sum(cropdata$intercrop == 1, na.rm = TRUE)
[1] 3752

There are 1233 unique households that have at least one intercropped plot, and in total, these households have 3752 intercropped plots in Long rainy season wave 4

# Focus on households that have intercropped plots
intercrop_data <- cropdata |> 
  filter(intercrop == 1) |> 
  select(y4_hhid, plotnum, zaocode, zaoname)

# Table of intercropped plots by zaocode

Maize is the most frequently intercropped crop (1432 plots), followed by beans (531) and groundnut (306) in Long rainy season wave 4.

# i) Plots and households where either maize or a legume crop is intercropped
maize_or_leg_intercrop_data <- intercrop_data |>  
  filter(zaocode %in% c(11, 31, 32, 33, 34, 35, 36, 37, 43, 47))

# Count the total number of intercrops in plots where either maize or a legume is intercropped
num_plots_maize_or_leg <- nrow(maize_or_leg_intercrop_data)

# Number of households where at least one plot has maize or legume intercropped
num_HH_maize_or_leg <- maize_or_leg_intercrop_data |> 
  distinct(y4_hhid) |> 
  nrow()

# ii) Plots and households where both maize and at least one legume crop are intercropped
# Group by household and plot, and check for the presence of maize and legumes
maize_and_leg_intercrop_data <- maize_or_leg_intercrop_data |> 
  group_by(y4_hhid, plotnum) |> 
  summarise(
    has_maize = any(zaocode == 11),         # Check if maize is intercropped
    num_legumes = sum(zaocode != 11),       # Count how many legumes are intercropped
    .groups = "drop"
  ) |> 
  filter(has_maize & num_legumes > 0)  # Filter for plots with both maize and legumes

# Count the total number of intercrops in plots where both maize and legume intercropped
num_plots_maize_and_leg <- nrow(maize_and_leg_intercrop_data)

# Number of households with at least one plot with both maize and legume intercropped
num_HH_maize_and_leg <- maize_and_leg_intercrop_data |> 
  distinct(y4_hhid) |> 
  nrow()
